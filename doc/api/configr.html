Built-in Configr plugin API functions (getEnv, getVariable, logInfo, fileExists, etc.)

getEnv name
Gets the value of an environment variable.
NameTypeDescription
name string The name of the environment variable.
Returns: string|nil
local home = getEnv("HOME")
getVariable name
Gets a variable from the current configuration context.
NameTypeDescription
name string The variable name to look up.
Returns: any
local msg = getVariable("message")
setVariable name value
Sets a variable in the global configuration context.
NameTypeDescription
name string The variable name.
value any The value to assign.
Returns: nil
setVariable("notify_message", "hello")
expandVariables str
Expands Configr variable references ($var) in a string.
NameTypeDescription
str string String containing $variable references.
Returns: string
local val = expandVariables("prefix_$some_var")
logInfo message
Logs an info-level message to the Configr logger.
NameTypeDescription
message string The message to log.
Returns: nil
logInfo("Block executed successfully")
logWarning message
Logs a warning-level message to the Configr logger.
NameTypeDescription
message string The message to log.
Returns: nil
logWarning("Deprecated option used")
logError message
Logs an error-level message to the Configr logger.
NameTypeDescription
message string The message to log.
Returns: nil
logError("Failed to apply block")
logDebug message
Logs a debug-level message to the Configr logger.
NameTypeDescription
message string The message to log.
Returns: nil
logDebug("Variable value: " .. val)
fileExists path
Checks whether a file exists at the given path.
NameTypeDescription
path string Path to check.
Returns: boolean
if fileExists("/tmp/flag") then ... end
readFile path
Reads the entire contents of a file as a string.
NameTypeDescription
path string Path to the file.
Returns: string
local data = readFile("/etc/config.cfg")
writeFile path content
Writes string content to a file, replacing existing content.
NameTypeDescription
path string Destination file path.
content string Content to write.
Returns: nil
writeFile("/tmp/out.txt", "hello")
appendFile path content
Appends string content to the end of a file.
NameTypeDescription
path string Destination file path.
content string Content to append.
Returns: nil
appendFile("/tmp/log.csv", "entry,data\n")
runCommand command
Runs a shell command through Configr’s process backend.
NameTypeDescription
command string Command string to execute.
Returns: integer
local code = runCommand("touch /tmp/done")
configrCacheDir
Returns the Configr cache directory path.
Returns: string
local dir = configrCacheDir()
configrBackupDir
Returns the Configr backup directory path.
Returns: string
local dir = configrBackupDir()
emitStatusUpdate moduleId level message
Emits a status update event to the Configr event bus.
NameTypeDescription
moduleId string The module or block identifier.
level string Severity level: "info", "warning", "error", or "debug".
message string The status message.
Returns: nil
emitStatusUpdate(block.id, "info", "Applying block " .. block.id)
registerBlock blockType callbacks
Registers a Lua-defined action block type with callbacks.
NameTypeDescription
blockType string The block type name (e.g. "notify").
callbacks registerBlock.callbacks Callback table. See "registerBlock.callbacks" table schema for field types.
Returns: nil
registerBlock("notify", {
  execute = function(block)
    logInfo("Running block: " .. block.id)
  end,
  rollback = function(block)
    logInfo("Rolling back: " .. block.id)
  end
})
getContext key
Returns a value from the default plugin context table by key.
NameTypeDescription
key string Context key ("platform", "architecture", "hostname", "os" (table), "user" (table), "env" (table), "configr" (table)).
Returns: any
local platform = getContext("platform")
local distro = getContext("os")["distribution"]

Assertion functions for sweep-test fixture scripts.

assertFileExists path
Asserts that a file exists.
NameTypeDescription
path string Path to the file.
Returns: nil
assertFileExists('/tmp/myfile.txt')
assertFileNotExists path
Asserts that a file does not exist.
NameTypeDescription
path string Path to the file.
Returns: nil
assertFileNotExists('/tmp/deleted.txt')
assertFileContains path pattern
Asserts that a file contains the given substring.
NameTypeDescription
path string Path to the file.
pattern string Substring to search for.
Returns: nil
assertFileContains('/tmp/log.txt', 'success')
assertFileNotContains path pattern
Asserts that a file does not contain the given substring.
NameTypeDescription
path string Path to the file.
pattern string Substring to search for.
Returns: nil
assertFileNotContains('/tmp/log.txt', 'ERROR')
assertDirExists path
Asserts that a directory exists.
NameTypeDescription
path string Path to the directory.
Returns: nil
assertDirExists('/tmp/mydir')
assertDirNotExists path
Asserts that a directory does not exist.
NameTypeDescription
path string Path to the directory.
Returns: nil
assertDirNotExists('/tmp/deleted_dir')
makeDir path
Creates a directory recursively (like mkdir -p).
NameTypeDescription
path string Directory path to create.
Returns: nil
makeDir(tempDir() .. '/mydir')
removeTree path
Removes a directory tree (like rm -rf).
NameTypeDescription
path string Directory path to remove.
Returns: nil
removeTree(tempDir() .. '/mydir')
tempDir
Returns the OS temporary directory path.
Returns: string
local tmp = tempDir()
osName
Returns the OS name (linux, macos, windows, etc).
Returns: string
if osName() == 'windows' then ... end
